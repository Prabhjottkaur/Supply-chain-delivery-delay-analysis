SELECT order_id, COUNT(*)
FROM order_fulfillment
GROUP BY order_id
HAVING COUNT(*) > 1;

FROM order_fulfillment
WHERE shipping_date < order_date;

##Checking negative financial values

FROM order_fulfillment
WHERE sales < 0 
   OR order_item_total < 0

CREATE VIEW delivery_analysis AS
SELECT
    order_date,
    order_region,
    category_name,
    shipping_mode,
    customer_segment,
    delivery_status,
    late_delivery_risk,
    (days_shipping_real - days_shipment_scheduled) AS delivery_delay_days
FROM order_fulfillment;
-------


 Which regions have highest delay rates ##
SELECT
    order_region,
    COUNT(*) AS total_orders,
    AVG(delivery_delay_days) AS avg_delay,
    SUM(CASE WHEN delivery_status = 'Late Delivery' THEN 1 ELSE 0 END) * 100.0 / COUNT(*) AS late_delivery_rate
FROM delivery_analysis
GROUP BY order_region;



## Which shipping_mode have highest delay rates ##
SELECT
    shipping_mode,
    COUNT(*) AS total_orders,
    AVG(delivery_delay_days) AS avg_delay,
    SUM(CASE WHEN delivery_status = 'Late Delivery' THEN 1 ELSE 0 END) * 100.0 / COUNT(*) AS late_delivery_rate
FROM delivery_analysis
GROUP BY shipping_mode;


## Which category have highest delay rates ##
SELECT
    category_name,
    COUNT(*) AS total_orders,
    AVG(delivery_delay_days) AS avg_delay,
    SUM(CASE WHEN delivery_status = 'Late Delivery' THEN 1 ELSE 0 END) * 100.0 / COUNT(*) AS late_delivery_rate
FROM delivery_analysis
GROUP BY category_name;


## Which order region and shipping mode are causing the delay rates ##
SELECT
    order_region, shipping_mode,
    COUNT(*) AS total_orders,
    AVG(delivery_delay_days) AS avg_delay,
    SUM(CASE WHEN delivery_status = 'Late Delivery' THEN 1 ELSE 0 END) * 100.0 / COUNT(*) AS late_delivery_rate
FROM delivery_analysis
GROUP BY   order_region, shipping_mode;


## Are delivery delay increasing overtime ##
SELECT
    DATE(order_date),
    AVG(delivery_delay_days) AS avg_delay,
    SUM(CASE WHEN delivery_status = 'Late Delivery' THEN 1 ELSE 0 END) * 100.0 / COUNT(*) AS late_delivery_rate
FROM delivery_analysis
GROUP BY DATE(order_date);


## Do order spike increases delivery delay ##
SELECT
    DATE(order_date),
    COUNT(*) AS order_volume,
    AVG(delivery_delay_days) AS avg_delay,
    SUM(CASE WHEN delivery_status = 'Late Delivery' THEN 1 ELSE 0 END) * 100.0 / COUNT(*) AS late_delivery_rate
FROM delivery_analysis
GROUP BY DATE(order_date);

## Which customer segment faces more delays##
SELECT
    customer_segment,
    AVG(delivery_delay_days) AS avg_delay,
    SUM(CASE WHEN delivery_status = 'Late Delivery' THEN 1 ELSE 0 END) * 100.0 / COUNT(*) AS late_delivery_rate
FROM delivery_analysis
GROUP BY customer_segment;
